规则 1：相邻短卡自动合列

buildStandardSlots()

按配置顺序扫描第二层卡片：只要相邻两张卡片中至少一张 ≤ 2 项、且两张合计 ≤ 4 项，就上下叠放进同一个槽位（下方卡片加 flex-1 自动撑齐整行高度），既保留配置顺序，又避免短卡旁大片留白。 

function buildStandardSlots(categories) {
  const slots = [];
  for (let i = 0; i < categories.length; i++) {
    const cur = categories[i];
    const next = categories[i + 1];
    const curLen = (cur.items || []).length;
    const nextLen = next ? (next.items || []).length : Infinity;

    // 相邻短卡配对条件：至少一张 <= 2 项 且 合计 <= 4 项
    if (next && Math.min(curLen, nextLen) <= 2 && (curLen + nextLen) <= 4) {
      slots.push({ type: 'pair', items: [cur, next] });
      i++; // 跳过已合列的下一张
    } else {
      slots.push({ type: 'single', items: [cur] });
    }
  }
  return slots;
}

规则 2：按槽位数自动铺满 6 列

n % 3 满格算法

把桌面端一行等分为 6 列（lg:grid-cols-6）。根据合列后的槽位总数 n 分配列宽：除 n = 1 外，绝不出现单卡独占一行或末尾空列（如 7 槽位排成 3 + 2 + 2）。 

const n = slots.length;
const standardHtml = slots.map((slot, idx) => {
  let spanClass = 'lg:col-span-2'; // 默认 1/3 宽（每行 3 槽位）

  if (n === 1) {
    spanClass = 'lg:col-span-6';   // 仅 1 槽位：占满整行
  } else if (n % 3 === 2 && idx >= n - 2) {
    spanClass = 'lg:col-span-3';   // 余 2：末 2 槽位各占 1/2 宽
  } else if (n % 3 === 1 && idx >= n - 4) {
    spanClass = 'lg:col-span-3';   // 余 1：末 4 槽位排成 2+2
  }

  return renderSlot(slot, spanClass);
}).join('');

// 渲染时标记超出上限的条目

  ? ' extra-item hidden' : '';

// 搜索时自动透出匹配到的隐藏条目，并隐藏空槽位
const isCollapsedExtra = !q && item.classList.contains('extra-item')
